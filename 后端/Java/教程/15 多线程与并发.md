### 锁
```java
// 共享资源  
private int count = 0;  
  
// 默认非公平锁；new ReentrantLock(true) 可开启公平锁  
private final Lock lock = new ReentrantLock();  
  
public void increment() {  
    lock.lock();          // 获取锁，拿不到就阻塞等待  
    try {  
        count++;  
        System.out.println(Thread.currentThread().getName()  
                + " count = " + count);  
    } finally {  
        lock.unlock();    // 必须在 finally 中释放，防止异常后死锁  
    }  
}
```

## 等待唤醒机制
### 生产者与消费者
```java
public class Cook extends Thread {  
    @Override  
    public void run() {  
        while (true) {  
            synchronized (Desk.lock) {  
                if (Desk.count == 0) {  
                    break;  
                } else {  
                    if (Desk.foodFlag == 0) {  
                        Desk.foodFlag = 1;  
                        System.out.println("做了一碗面");  
                        Desk.lock.notifyAll();  
                    } else {  
                        try {  
                            Desk.lock.wait();  
                        } catch (InterruptedException e) {  
                            throw new RuntimeException(e);  
                        }  
                    }                }            }        }    }}

public class Desk {  
    public static int foodFlag = 0;  
    public static  int count =10;  
    public static  Object lock = new Object();  
}

public class Foodie extends Thread {

    public void run() {  
        while (true) {  
            synchronized (Desk.lock) {  
                if (Desk.count == 0) {  
                    break;  
                } else {  
                    if (Desk.foodFlag == 0) {  
                        try {  
                            Desk.lock.wait();  
                        } catch (InterruptedException e) {  
                            throw new RuntimeException(e);  
                        }

                        Desk.count--;

                        System.out.println("吃了一碗面条， 还剩" + Desk.count + "碗");  
                        Desk.lock.notifyAll();  
                    }  
                }            }        }    }}
```

### 阻塞队列实现
![[Pasted image 20260911172929.png]]

```java
public class ThreadDemo {  
    static void main() {  
        ArrayBlockingQueue<String> queue = new ArrayBlockingQueue<>(1);  
  
        Cook cook = new Cook(queue);  
        Foodie foodie = new Foodie(queue);  
  
        cook.setName("厨师");  
        foodie.setName("食客");  
  
        cook.start();  
        foodie.start();  
  
    }  
}

public class Foodie extends Thread {  
  
    ArrayBlockingQueue<String> queue;  
  
    public Foodie(ArrayBlockingQueue<String> queue) {  
        this.queue = queue;  
    }  
  
    @Override  
    public void run() {  
        while (true) {  
            try {  
                String foodies = queue.take();  
                System.out.println("食客吃了面条一碗");  
            } catch (InterruptedException e) {  
                throw new RuntimeException(e);  
            }  
        }    }}

public class Cook extends Thread {  
    ArrayBlockingQueue<String> queue;  
  
    public Cook (ArrayBlockingQueue<String> queue){  
        this.queue = queue;  
    }  
  
    @Override  
    public void run() {  
        while (true) {  
            try {  
                queue.put("面条");  
            } catch (InterruptedException e) {  
                throw new RuntimeException(e);  
            }  
            System.out.println("厨师放了面条一碗");  
        }  
    }}
```
