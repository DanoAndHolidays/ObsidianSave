JWT令牌生成及校验
	Jwts.builder()...
	Jwts.parser()...

JWT令牌解析(校验）时什么情况会报错?
	JWT令牌被篡改 或过期失效了

JWT校验时使用的签名秘钥，必须和生成JWT令牌时使用的秘钥是配套的

可供参考的实现：
```java
package org.dano.utils;

import io.jsonwebtoken.Claims;
import io.jsonwebtoken.JwtException;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.security.Keys;

import javax.crypto.SecretKey;
import java.nio.charset.StandardCharsets;
import java.time.Duration;
import java.time.Instant;
import java.util.Date;
import java.util.Map;

/**
 * JWT 工具类，提供令牌构建、解析和有效性校验功能。
 */
public final class JwtUtils {

    /**
     * 默认密钥仅适合本地开发。生产环境应通过环境变量 JWT_SECRET 或
     * JVM 参数 -Djwt.secret=xxx 设置至少 32 字节的密钥。
     */
    private static final String DEFAULT_SECRET =
            "tlias-web-management-jwt-secret-key-2026";

    private static final Duration DEFAULT_EXPIRATION = Duration.ofHours(12);
    private static final SecretKey SECRET_KEY = Keys.hmacShaKeyFor(
            resolveSecret().getBytes(StandardCharsets.UTF_8)
    );

    private JwtUtils() {
    }
    /**
     * 使用默认的 12 小时有效期构建 JWT。
     */
    public static String buildToken(Map<String, ?> claims) {
        return buildToken(claims, DEFAULT_EXPIRATION);
    }

    /**
     * 按指定的毫秒数构建 JWT。
     *
     * @param claims           JWT 中保存的自定义数据
     * @param expirationMillis 从当前时间开始计算的有效时长（毫秒）
     */
    public static String buildToken(Map<String, ?> claims, long expirationMillis) {
        if (expirationMillis <= 0) {
            throw new IllegalArgumentException("JWT 过期时间必须大于 0 毫秒");
        }
        return buildToken(claims, Duration.ofMillis(expirationMillis));
    }

    /**
     * 按指定的 Duration 构建 JWT。
     *
     * @param claims     JWT 中保存的自定义数据
     * @param expiration 从当前时间开始计算的有效时长
     */
    public static String buildToken(Map<String, ?> claims, Duration expiration) {
        if (claims == null) {
            throw new IllegalArgumentException("JWT claims 不能为空");
        }
        if (expiration == null || expiration.isZero() || expiration.isNegative()) {
            throw new IllegalArgumentException("JWT 过期时间必须大于 0");
        }

        Instant issuedAt = Instant.now();
        Instant expiresAt = issuedAt.plus(expiration);

        return Jwts.builder()
                .claims(claims)
                .issuedAt(Date.from(issuedAt))
                .expiration(Date.from(expiresAt))
                .signWith(SECRET_KEY)
                .compact();
    }

    /**
     * 解析并校验 JWT 的签名和过期时间。
     * 校验失败时，JJWT 会抛出 JwtException 的具体子类。
     */
    public static Claims parseToken(String token) {
        return Jwts.parser()
                .verifyWith(SECRET_KEY)
                .build()
                .parseSignedClaims(normalizeToken(token))
                .getPayload();
    }

    /**
     * 获取一个指定类型的声明值。
     */
    public static <T> T getClaim(String token, String claimName, Class<T> requiredType) {
        if (claimName == null || claimName.isBlank()) {
            throw new IllegalArgumentException("claimName 不能为空");
        }
        if (requiredType == null) {
            throw new IllegalArgumentException("requiredType 不能为空");
        }
        return parseToken(token).get(claimName, requiredType);
    }

    /**
     * 判断 JWT 是否有效。签名错误、格式错误或已过期都会返回 false。
     */
    public static boolean isValid(String token) {
        try {
            parseToken(token);
            return true;
        } catch (JwtException | IllegalArgumentException exception) {
            return false;
        }
    }
    private static String normalizeToken(String token) {
        if (token == null || token.isBlank()) {
            throw new IllegalArgumentException("JWT token 不能为空");
        }

        String normalized = token.trim();
        if (normalized.regionMatches(true, 0, "Bearer ", 0, 7)) {
            normalized = normalized.substring(7).trim();
        }
        if (normalized.isEmpty()) {
            throw new IllegalArgumentException("JWT token 不能为空");
        }
        return normalized;
    }

    private static String resolveSecret() {
        String secret = System.getProperty("jwt.secret");
        if (secret == null || secret.isBlank()) {
            secret = System.getenv("JWT_SECRET");
        }
        if (secret == null || secret.isBlank()) {
            secret = DEFAULT_SECRET;
        }
        if (secret.getBytes(StandardCharsets.UTF_8).length < 32) {
            throw new IllegalStateException("JWT 密钥不能少于 32 字节");
        }
        return secret;
    }
}
```

## Filter
概念 Filter 过滤器，是 JavaWeb 三大组件(Servlet、Filter、Listener)之一,过滤器可以把对资源的请求拦截下来，从而实现一些特殊的功能

过滤器一般完成一些通用的操作，比如：登录校验、统一编码处理、敏感字符处理等：
![[Pasted image 20260926191105.png]]

```java

        log.info("拦截开始，{}{}", servletRequest, servletRequest);

    @Override

        //Filter.super.destroy();

}
```

入门实现：
```java
@Slf4j
@WebFilter(urlPatterns = "/*")
public class TokenFilter implements Filter {
    @Override
    public void doFilter(ServletRequest servletRequest, ServletResponse servletResponse, FilterChain filterChain) throws IOException, ServletException {
        HttpServletRequest httpServletRequest = (HttpServletRequest) servletRequest;
        HttpServletResponse httpServletResponse = (HttpServletResponse) servletResponse;
        String uri = httpServletRequest.getRequestURI();

        if (uri.contains("/login")) {
            log.info("登录，直接放行");
            filterChain.doFilter(servletRequest, servletResponse);
        }

        String token = httpServletRequest.getHeader("token");

        if (token == null || token.isEmpty()) {
            httpServletResponse.setStatus(HttpServletResponse.SC_UNAUTHORIZED);
            log.info("未有合法的Token1");
            return;
        }

        try {
            JwtUtils.parseToken(token);
        } catch (Exception e) {
            httpServletResponse.setStatus(HttpServletResponse.SC_UNAUTHORIZED);
            log.info("未有合法的Token2");
            return;
        }

        log.info("Token有效，放行");
        filterChain.doFilter(servletRequest, servletResponse);
    }
}
```

```java
@Slf4j
@Component
public class TokenInterceptor implements HandlerInterceptor {
    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) throws Exception {

        String uri = request.getRequestURI();

            log.info("登录，直接放行");
            return true;

        if (token == null || token.isEmpty()) {
            response.setStatus(HttpServletResponse.SC_UNAUTHORIZED);
            log.info("未有合法的Token1");
            return false;

        try {
            JwtUtils.parseToken(token);
        } catch (Exception e) {

            log.info("未有合法的Token2");
            return false;
        }

        log.info("Token有效，放行");
        return true;
    }
}
```
