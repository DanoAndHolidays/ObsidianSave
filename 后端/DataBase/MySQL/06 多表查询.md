## 子查询
![[Pasted image 20260920002255.png]]

```sql
-- 标量
SELECT MIN(entry_date)
FROM emp

SELECT *
FROM emp
WHERE entry_date = '2018-06-01'

SELECT *
FROM emp
WHERE entry_date = (
  SELECT MIN(entry_date)
  FROM emp
)
```

```sql
-- 列
SELECT dept.id
FROM dept
WHERE name = '小卖部' OR name = '火箭部'

SELECT *
FROM emp
WHERE dept_id in (7, 9)

SELECT *
FROM emp
WHERE dept_id in (
  SELECT dept.id FROM dept WHERE name = '小卖部' OR name = '火箭部'
)


-- 行
SELECT emp.salary, emp.job FROM emp WHERE name = '张无忌'
SELECT * FROM emp WHERE (salary, job) = (8500, 1)
SELECT * FROM emp WHERE (salary, job) = (SELECT emp.salary, emp.job FROM emp WHERE name = '张无忌')


-- 表
SELECT dept_id, MAX(salary) FROM emp GROUP BY dept_id;

-- 将结果看作表，进行内连接查询
SELECT *
FROM emp, (SELECT dept_id, MAX(salary) as max_salary FROM emp GROUP BY dept_id) as a
WHERE emp.dept_id = a.dept_id AND emp.salary = a.max_salary
```

我自己写的，不一定最优，这些其实都可以先内连接的，我都是先子查询

```sql
SELECT *
FROM emp
WHERE emp.dept_id = (SELECT dept.id FROM dept WHERE name = '火箭部')
AND emp.gender = 1
AND emp.entry_date > '2011-05-01'

-- SELECT AVG(emp.salary) FROM emp
SELECT *
FROM emp
WHERE emp.salary < (
	SELECT AVG(emp.salary) FROM emp
) AND emp.gender = 1

SELECT *
FROM dept,
	(SELECT emp.dept_id, COUNT(emp.dept_id) AS num_dept FROM emp GROUP BY dept_id) AS a
WHERE dept.id = a.dept_id
AND num_dept > 2

SELECT *
FROM emp
WHERE emp.salary > 10000
AND emp.dept_id = (SELECT dept.id FROM dept WHERE dept.name = '火箭部')
AND emp.entry_date > '2010-05-01'
ORDER BY emp.salary DESC

SELECT *
FROM emp , (
	SELECT emp.dept_id, AVG(emp.salary) AS avg_salary
	FROM emp
	GROUP BY emp.dept_id
) AS a
WHERE emp.dept_id = a.dept_id AND emp.salary < avg_salary
```
