# 竞态问题
在多个异步操作中，我们只关心最新一次操作的结果，而之前操作的结果可能会覆盖新的结果。实际的例子是在连续发送请求时，由于网络问题，先发送的请求较晚返回，覆盖了较新的请求。

一个简单的input竞态问题解决方案：
```javascript
// 1. 标准的防抖函数（保持不变）
function debounce(func, time) {
    let timer = null;
    return function (...args) {
        if (timer) clearTimeout(timer);
        timer = setTimeout(() => {
            func.apply(this, args);
        }, time);
    };
}

// 2. 维护一个全局的 AbortController 实例
let currentController = null;

async function fetchSuggest(keyword) {
    // 【核心取消逻辑】：在发起新请求前，检查是否有正在进行的旧请求。
    // 如果有，直接调用 abort() 把它掐掉！
    if (currentController) {
        currentController.abort();
    }
    
    // 创建一个新的 AbortController，用于控制当前这次请求
    currentController = new AbortController();
    const signal = currentController.signal;

    try {
        // 模拟一个带有随机延迟的网络请求（将 signal 传入 fetch）
        const mockFetch = (kw, signal) => {
            const delay = Math.random() * 1000;
            return new Promise((resolve, reject) => {
                const timer = setTimeout(() => {
                    resolve(`这是 "${kw}" 的搜索结果`);
                }, delay);
                
                // 监听取消信号，如果请求被取消，要清除定时器并抛出错误
                signal.addEventListener('abort', () => {
                    clearTimeout(timer);
                    reject(new DOMException('请求已取消', 'AbortError'));
                });
            });
        };

        // 将 signal 传入真实的 fetch 或模拟请求中
        const res = await mockFetch(keyword, signal);
        
        // 只有没有被取消的请求，才会走到这里并更新 UI
        console.log("✅ 最终渲染结果：", res);
        
    } catch (error) {
        // 【核心错误处理】：区分是“主动取消”还是“真正的网络报错”
        // 如果是 AbortError，说明是我们自己取消的，直接静默忽略即可
        if (error.name === 'AbortError') {
            console.log("❌ 请求被取消，忽略该结果：", keyword);
            return;
        }
        console.error('真正的网络请求出错:', error);
    }
}

// 3. 绑定到 input 输入框
const debouncedFetch = debounce(fetchSuggest, 300);

document.querySelector('input').addEventListener('input', (e) => {
    const keyword = e.target.value;
    if (keyword.trim()) {
        debouncedFetch(keyword);
    }
});
```
