# 2267 检查是否有合法括号字符串路径
> Last Format Time：10/6/2026 02:27:25

[https://leetcode.cn/problems/check-if-there-is-a-valid-parentheses-string-path](https://leetcode.cn/problems/check-if-there-is-a-valid-parentheses-string-path)

```java
class Solution {
    public boolean hasValidPath(char[][] grid) {
        int m = grid.length, n = grid[0].length;
        // 这里已经避免了越界的问题
        boolean[][][] f = new boolean[m + 1][n + 1][m + n + 2];
        
        if (grid[0][0] == '(') {
            f[1][1][1] = true;
        } else {
            return false;
        }
        
        for (int i = 1; i <= m; i++) {
            for (int j = 1; j <= n; j++) {
            
                boolean isLeft = grid[i - 1][j - 1] == '(';
                for (int k = 0; k <= m + n; k++) {
                    if (isLeft) {
                        if (k == 0) continue;
                        f[i][j][k] |= f[i - 1][j][k - 1];
                        f[i][j][k] |= f[i][j - 1][k - 1];
                    } else {
                        f[i][j][k] |= f[i - 1][j][k + 1];
                        f[i][j][k] |= f[i][j - 1][k + 1];
                    }
                }
                
            }
        }
        return f[m][n][0];
    }

}
```
