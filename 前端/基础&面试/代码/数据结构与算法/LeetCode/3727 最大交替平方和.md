class Solution {
    public long maxAlternatingSum(int[] nums) {
        // 这道题使用排序，按就绝对值的大小，从大到小
        // 大的在前面加，后面小的减去

        int n = nums.length;

        int mid = n / 2 + n % 2;

        // int[] 转 Integer[]
        Integer[] arr = Arrays.stream(nums)
                .boxed()
                .toArray(Integer[]::new);

        Arrays.sort(arr, (a, b) -> Integer.compare(Math.abs(b), Math.abs(a)));

        System.out.println(mid + Arrays.toString(arr));

        long ans =0;

        for(int i = 0; i < n; i ++){
            if(i < mid){
                ans += arr[i] * arr[i];
            }else{
                ans -= arr[i] * arr[i];
            }
            
        }

        return ans;
    }
}

class Solution {
    public long maxAlternatingSum(int[] nums) {
        int n = nums.length;
        for (int i = 0; i < n; i++) {
            nums[i] *= nums[i];
        }
        Arrays.sort(nums);

        int m = n / 2;
        // 交替和：减去小的，加上大的
        long ans = 0;
        for (int i = 0; i < m; i++) {
            ans -= nums[i];
        }
        for (int i = m; i < n; i++) {
            ans += nums[i];
        }
        return ans;
    }
}
