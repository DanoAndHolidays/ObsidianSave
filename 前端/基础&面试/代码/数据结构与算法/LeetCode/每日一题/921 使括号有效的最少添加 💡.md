```java
class Solution {
    public int minAddToMakeValid(String s) {
        int ans = 0;
        int blance = 0;

        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);

            if (c == '(') {
                blance++;
            } else {
                blance--;
            }

            if(blance < 0){
                blance = 0;
                ans += 1;
            }
        }

        return blance + ans;
    }
}
```
