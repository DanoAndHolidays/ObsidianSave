class Solution {
    public int maxSum(List<Integer> nums, int k) {
        // 将一个元素的部分位转移到另一个元素
        // 要使平方和最大，贪心地让每个数尽可能大
        // 相当于把 1 都聚集在一个数中，比分散到不同的数更好


        final long MOD = 1_000_000_007;
        int[] cnt = new int[30];
        for (int x : nums) {
            for (int i = 0; i < 30; i++) {
                cnt[i] += (x >> i) & 1;
            }
        }
        long ans = 0;
        while (k-- > 0) {
            int x = 0;
            for (int i = 0; i < 30; i++) {
                if (cnt[i] > 0) {
                    cnt[i]--;
                    x |= 1 << i;
                }
            }
            ans = (ans + (long) x * x) % MOD;
        }
        return (int) ans;
    }
}
