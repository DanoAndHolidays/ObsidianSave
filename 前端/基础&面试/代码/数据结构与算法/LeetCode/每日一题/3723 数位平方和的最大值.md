[3723. 数位平方和的最大值](https://leetcode.cn/problems/maximize-sum-of-squares-of-digits/)

#中等

```ts
function maxSumOfSquares(num: number, sum: number): string {
    // 要想各个位上的数字的平方和最大，那么就尽可能的使每个位都达到9，使数字集中
    // 比如 30 肯定好于 12 21，因为他数值都集中于3中
    // 由于每个位最多0-9，所以sum <= num * 9
    let ans = "";
    if (sum > num * 9) return ans;

    let numOfNine = Math.floor(sum / 9);
    let hasNum = sum % 9 !== 0 ? 1 : 0
    let otherNum: Number | String = 0
    if(hasNum === 1) {
        otherNum = sum % 9
    }else{
        otherNum = ""
    }

    ans = "9".repeat(numOfNine) + otherNum + "0".repeat(num - numOfNine - hasNum)
    return ans
};
```
