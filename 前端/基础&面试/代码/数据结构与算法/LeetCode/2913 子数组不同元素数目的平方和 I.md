*暴力是简单的，但是树状数组是困难的，以后在说吧*
```ts
function sumCounts(nums: number[]): number {
    const n = nums.length;
    let ans = 0;
    for (let i = 0; i <= n; i++) {
        for (let j = i + 1; j <= n; j++) {
            const set = new Set(nums.slice(i, j))
            // console.log(set, i, j)
            ans += set.size * set.size
        }
    }

    return ans;
};
```
