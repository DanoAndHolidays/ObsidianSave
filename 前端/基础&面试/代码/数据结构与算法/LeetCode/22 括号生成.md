# 22 括号生成
> Last Format Time：10/6/2026 02:27:25

[https://leetcode.cn/problems/generate-parentheses](https://leetcode.cn/problems/generate-parentheses)

```ts
function generateParenthesis(n: number): string[] {
    const m = n * 2;
    const path = Array(m)
    const ans = [];

    function dfs(i, blance) {
        if (i === m) {
            ans.push(path.join(""));
            return
        }

        if (blance < n) {
            path[i] = "(";
            dfs(i + 1, blance + 1)
        }
```

    }

    dfs(0, 0)

    return ans;
};
