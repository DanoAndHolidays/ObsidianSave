function maxNumOfSubstrings(s: string): string[] {
    // 注意，任何两个有效子串不可能重叠，除非其中一个包含另一个，一旦包含，被包含的肯定是更优解，因为他更短，所以可以贪心
    // 首先找出每个字符的起始和结束下标，若范围内存在起始索引更小或结束索引更大的其他字符，就扩张，最后扩展为一个子字符串
    // 按长度对所有有效子串进行排序，贪心地选取长度最小的子串，并舍弃与已选子串重叠的子串


    const n = s.length
    const map = new Map()
    for (let i = 0; i < n; i++) {
        const char = s[i]
        if (map.get(char)) {
            map.set(char, { ...map.get(char), end: i })
        } else {
            map.set(char, { start: i, end: i })
        }
    }
    const arr = [...map.values()].map(({ start, end }) => [start, end]).sort((a, b) => a[0] - b[0])

    // console.log(map)
    let intervals = []
    for (let i = 0; i < arr.length; i++) {
        const interval = getInterval(arr[i][0], arr[i][1])
        if (interval) intervals.push(interval)
    }

    intervals.sort((a, b) => (a[1] - a[0]) - (b[1] - b[0]))

    function getInterval(start, end) {
        let newEnd = end
        // console.log(start, end)
        for (let i = start; i <= newEnd; i++) {
            const ch = s[i]
            // 当前区间包含 ch，但 ch 在 start 左边还有出现
            // 当前 start 无法构成合法最小区间
            if (map.get(ch).start < start) {
                return null
            }
            newEnd = Math.max(newEnd, map.get(ch).end)
        }
        return [start, newEnd]
    }
    console.log(intervals)
    const ans: number[][] = []

    for (const interval of intervals) {
	    // 方法测试数组中是否至少有一个元素通过了由提供的函数实现的测试
	    
        const overlap = ans.some(
            selected =>
                Math.max(selected[0], interval[0]) <=
                Math.min(selected[1], interval[1])
        )

        if (!overlap) {
            ans.push(interval)
        }
    }

    return ans.map((interval) => s.slice(interval[0], interval[1] + 1))
};
