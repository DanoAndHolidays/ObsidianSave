/* 减少你掉头发的概率 */
* {
  margin: 0;
  padding: 0;
  box-sizing: border-box;
}

html {
  background-color: brown;
}

.container {
  width: 400px;
  background-color: rgb(251, 251, 251);
  margin: auto;
  border-radius: 15px;
  padding: 30px;
}

.container h1 {
  text-align: center;
  margin-bottom: 30px;
}

.form-item {
  margin-bottom: 10px;
}

/* 浏览器的默认样式覆盖 */
.txt {
  outline: none;
  border: 1px solid red;
  width: 100%;
  height: 40px;
  border-radius: 4px;
  font-size: 10px;
  padding: 0 10px;
}

.txt:focus {
  border-color: blue;
}

select.txt {
  height: 100px;
}

textarea.txt {
  resize: none;
}

button {
  width: 100px;
  height: 40px;
  border: none;
  outline: none;
  border-radius: 5px;
  cursor: pointer;
  background-color: rgb(177, 0, 0);
  color: white;
}

button:hover {
  background-color: rgb(121, 0, 0);
}

/* 使用伪类实现禁用的按钮样式 */
button:disabled {
  background-color: rgb(90, 96, 96);
  cursor: not-allowed;
}

/* 浮动，狗屎 解决高度坍塌的问题，CSS3以后就几乎不使用了*/
.left {
  float: left;
}
.right {
  float: right;
}

.clearfix::after {
  content: '';
  display: block;
  clear: both;
}

.cap.txt {
  width: 230px;
}

/* 针对行盒不对齐的情况可以使用vertical-align */
.policy {
  font-size: 12px;
  vertical-align: 2.5px;
}

label{
  cursor: pointer;
}
