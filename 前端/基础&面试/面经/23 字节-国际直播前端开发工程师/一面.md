二叉树右侧的视角：
我的实现

```js
// root = [1, 2, 3, null, 5, null, 4]

const fn = (root) => {
    if (!root) return []
    let res = []
    let curVals = [root]

    do {
        let nxtVals = []
        curVals.forEach((node) => {
            if (node.left) nxtVals.push(root.left.val)
            if (node.right) nxtVals.push(root.right.val)
        })

        const val = curVals[curVals.length]
        res.push(val)
        curVals = nxtVals
    } while (nxtVals.length > 0)

    return res
}

```

标准实现：
```js
const fn = (root) => {
    if (!root) return []

    const res = []
    let curVals = [root]

    while (curVals.length > 0) {
        // 当前层最右边
        res.push(curVals[curVals.length - 1].val)

        const nxtVals = []

        curVals.forEach((node) => {
            if (node.left) nxtVals.push(node.left)
            if (node.right) nxtVals.push(node.right)
        })

        curVals = nxtVals
    }

    return res
}
```
