- 闭包：这道题目是在一个数组中存了很多匿名函数，匿名函数在for循环中访问外部的闭包的
```js
function func() {
    var arr = [];
    // var i 是个闭包
	
	
    // var i = 0 
    // for(; i < 3; i++ ) { ... } 这样写也可以
    for (var i = 0; i < 3; i++) {
        arr.push(() => {
            console.log(i);
        });
    }
    return arr;
}

var result = func();
result.forEach((item) => {
    item(); // 期望输出 0, 1, 2，但实际打印了三个 3
});
```

- 最长连续重复字符的子串
```js
function fn(str) {
    let pre = 0
    let cur = 1
    let n = str.length

    let ans = Array(n).fill(0)
    
    if (n === 0) return { subStr: '', pos: 0 }
    if (n === 1) return str

    for (let i = 1; i < n; i++) {
        if (str[cur] === str[pre]) {
            cur++
            ans[pre] = cur - pre + 1
        } else {
            cur++
            pre++
        }

    }

    let maxValue = Math.max(...ans)
    let maxIndex = ans.indexOf(maxValue)

    return { subStr: str.slice(maxIndex, maxIndex + maxValue - 1 < 0 ? 0 : maxIndex + maxValue - 1), pos: maxIndex }
}
```
